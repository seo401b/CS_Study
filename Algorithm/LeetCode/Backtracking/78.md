class Solution {

private:
    std::vector<std::vector<int>> result;
    std::vector<int> cur;
    void dfs(int mask, std::vector<int>& nums){
        if(mask>= (1 << nums.size())) return;

        for(int i = 0; i < nums.size(); ++i){
            if(mask&(1 << i)){
                cur.push_back(nums[i]);
            }
        }
        result.push_back(cur);
        dfs(mask << 1, nums);
    }
public:
    std::vector<std::vector<int>> subsets(std::vector<int>& nums) {
        
        dfs(0, nums);
        return result;
    }
};

dfs 호출 시 0을 mask로 시작했는데 0은 아무리 밀어도 0이기때문에 무한 재귀를 돌게된다.
게다가 mask << 1을 하면 011 101 등의 이진수를 만들 수 없다.
십진수로 계속 +1을 하면 이진수 관점으로는 001 010 011 이런식으로 증가하기 때문에 dfs(mask+1, nums)가 더 부합한다.

class Solution {

private:
    std::vector<std::vector<int>> result;
    std::vector<int> cur;
    void dfs(int idx, std::vector<int>& nums){
        if(idx==nums.size()) {
            result.push_back(cur);
            return;
        }
        
        cur.push_back(nums[idx]);
        dfs(idx+1, nums);
        cur.pop_back();

        dfs(idx+1, nums);
    }
public:
    std::vector<std::vector<int>> subsets(std::vector<int>& nums) {
        
        dfs(0, nums);
        return result;
    }
};
