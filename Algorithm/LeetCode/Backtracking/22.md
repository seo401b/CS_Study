#include <iostream>
#include <vector>

class Solution {
private:
    std::vector<std::string> parenthesis {
        "(", ")"
    };
    void dfs(const int n, std::string current, int left, int right, std::vector<std::string>& result){
        if(left > n || right > n || left < right) return;
        else if(left==n && right==n) {
            result.push_back(current);
            return;
        }

        dfs(n, current.append(parenthesis[0]), left+1, right, result);
        dfs(n, current.append(parenthesis[1]), left, right+1, result);
    }
public:
    std::vector<std::string> generateParenthesis(int n) {
        std::vector<std::string> result;

        dfs(n, "(", 1, 0, result);
        return result;
    }
};

class Solution {
private:
    std::vector<std::string> parenthesis {
        "(", ")"
    };
    void dfs(const int n, int idx, std::string current, int cnt, std::vector<std::string>& result){
        if(cnt > n || cnt < 0) {
            current = "";
            return;
        }
        else if(idx==2*n && cnt==0) {
            result.push_back(current);
            return;
        }

        for(std::string& p: parenthesis){
            if(p=="("){
                dfs(n, idx+1, current.append(p), cnt+1, result);
            }
            else {
                dfs(n, idx+1, current.append(p), cnt-1, result);
            }
        }

    }
public:
    std::vector<std::string> generateParenthesis(int n) {
        std::vector<std::string> result;

        dfs(n, 1, "(", 1, result);
        return result;
    }
};

다음과 같이 변경했다. 위에서 케이스를 안나눠 문제가 발생했다 생각하고 for문을 돌려 괄호의 상태를 기준으로 케이스를 나눴는데 current의 상태를 고려하지 않고 돌아가는 for문이기에 잘못된 결과가 출력된다.
다음 시도에선 재귀함수 안에 하나의 재귀함수만 호출하며 괄호의 상태와 cnt로직을 바로 연동하여 하나의 변수로 다뤄보겠다.

class Solution {
private:
    std::vector<std::string> parenthesis {
        "(", ")"
    };
    void dfs(const int n, std::string current, int left, int right, std::vector<std::string>& result){
        if(current.length()==2*n) {
            result.push_back(current);
            return;
        }

        if(left < n){
            dfs(n, current+parenthesis[0], left+1, right, result);
        }

        if(left > right){
            dfs(n, current+parenthesis[1], left, right+1, result);
        }

    }
public:
    std::vector<std::string> generateParenthesis(int n) {
        std::vector<std::string> result;

        dfs(n, "(", 1, 0, result);
        return result;
    }
};
