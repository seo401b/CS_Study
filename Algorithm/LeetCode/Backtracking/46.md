    void dfs(int cur_idx){
        if(current_.size()==nums_.size()){
            result_.push_back(current_);
            return;
        }
         if(visited[cur_idx]){
             return;
         }
        for(int i = 0; i < nums_.size(); ++i){
            visited[i] = 1;
            current_.push_back(nums_[i]);
            dfs(i);
            current_.pop_back();
            visited[i] = 0;
        }
    }

    빈 배열 출력됨
    이유는 visited되었다면 함수 자체를 return하기 때문
    for문에서만 visited 로직이 놀아야 nums_배열의 요소 단위 제어가 가능할 듯
